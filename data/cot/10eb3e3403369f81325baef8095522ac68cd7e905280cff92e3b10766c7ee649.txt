   \text{Total naps} = 3 \text{ naps/week} \times 10 \text{ weeks} = 30 \text{ naps}

   \[
   \text{Total hours of naps} = 30 \text{ naps} \times 2 \text{ hours/nap} = 60 \text{ hours}
   \]
