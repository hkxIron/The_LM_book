     \text{Height increase needed} = \text{Target height} - \text{Current height} = 23 \text{ feet} - 5 \text{ feet} = 18 \text{ feet}

     \[
     \text{Number of years needed} = \frac{\text{Height increase needed}}{\text{Growth per year}} = \frac{18 \text{ feet}}{3 \text{ feet/year}} = 6 \text{ years}
     \]

     \[
     \text{Age when it reaches 23 feet} = \text{Current age} + \text{Number of years needed} = 1 \text{ year} + 6 \text{ years} = 7 \text{ years}
     \]
