     \[
     \text{Time} = \frac{\text{Distance}}{\text{Speed}}

     \[
     \text{Time uphill} = \frac{3 \text{ miles}}{2 \text{ MPH}} = 1.5 \text{ hours}

     \[

   \[
   \text{Total time} = \text{Time uphill} + \text{Time downhill} = 1.5 \text{ hours} + 0.6667 \text{ hours} \approx 2.1667 \text{ hours}
   \]

   \[
