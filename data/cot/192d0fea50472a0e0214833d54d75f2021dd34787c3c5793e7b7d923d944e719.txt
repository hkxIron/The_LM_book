     \text{Saturday's customers} = 3 \times 28 = 84 \text{ customers}

     \[
     \text{Saturday's tips} = 84 \text{ customers} \times 2.00 \text{ dollars/customer}

     \text{Saturday's tips} = 168 \text{ dollars}
     \]

     \[
     \text{Sunday's tips} = 36 \text{ customers} \times 2.00 \text{ dollars/customer}
     \]

     \text{Sunday's tips} = 72 \text{ dollars}
     \]

     \[
     \text{Total tips} = \text{Friday's tips} + \text{Saturday's tips} + \text{Sunday's tips}
     \]
     \[
