     \[
     \text{Profit} = 0.10 \times \text{Total Cost} = 0.10 \times 100
     \]

   \[
   \text{Profit} = 10

     \[
     \text{Total Revenue} = \text{Total Cost} + \text{Profit} = 100 + 10
     \]

   \[
   \text{Total Revenue} = 110
   \]

     \[
     \text{Price Per Bag} = \frac{\text{Total Revenue}}{\text{Number of Bags}} = \frac{110}{10}
