     \text{Ked's needs} = \frac{14}{2} = 7 \text{ eggs}

     \[
     \text{Total eggs for one week} = \text{Saly's eggs} + \text{Ben's eggs} + \text{Ked's eggs}

     \[
     = 10 + 14 + 7 = 31 \text{ eggs}
     \]

     \[
     \text{Total eggs for a month} = \text{Total eggs for one week} \times 4
     \]
     \[
     = 31 \times 4 = 124 \text{ eggs}
     \]
