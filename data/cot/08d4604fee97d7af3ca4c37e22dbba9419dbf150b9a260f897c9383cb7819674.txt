   - \[

   - \[
   12 \text{ stops} \times 20 \text{ minutes/stop} = 240 \text{ minutes}
   \]

   - \[
   \frac{240 \text{ minutes}}{60 \text{ minutes/hour}} = 4 \text{ hours}
   \]

   - \[
   14 \text{ hours} + 4 \text{ hours} = 18 \text{ hours}
   \]
