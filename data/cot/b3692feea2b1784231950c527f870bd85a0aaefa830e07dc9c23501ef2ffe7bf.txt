     \text{Revenue from children} = 28 \times 13

     \[
     28 \times 13 = 364
     \]

     \[
     \text{Total revenue} = \text{Revenue from adults} + \text{Revenue from children}
     \]

     \[
     \text{Total revenue} = 4758 + 364 = 5122
     \]
