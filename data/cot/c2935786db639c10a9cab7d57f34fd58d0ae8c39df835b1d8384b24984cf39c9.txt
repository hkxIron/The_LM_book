     \[
     \text{Total earnings} = \text{Earnings for the first month} + \text{Earnings for the second month} = 200 + 350 = 550
     \]

     \[
     \text{Total hours} = \frac{\text{Total earnings}}{\text{Hourly rate}} = \frac{550}{10} = 55
     \]
