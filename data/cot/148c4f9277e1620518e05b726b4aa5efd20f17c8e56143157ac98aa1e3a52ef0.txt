     \text{Profit per dozen} = \text{Revenue from selling} - \text{Cost of buying} = 12 \text{ dollars} - 2.40 \text{ dollars} = 9.60 \text{ dollars}

     \[
     \text{Number of dozens} = \frac{\text{Total amount to raise}}{\text{Profit per dozen}} = \frac{96 \text{ dollars}}{9.60 \text{ dollars}} = 10
