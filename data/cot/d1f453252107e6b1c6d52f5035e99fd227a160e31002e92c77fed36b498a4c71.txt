\[
\text{Total time for one outfit} = 2 \text{ (hat)} + 3 \text{ (scarf)} + 6 \text{ (sweater)} + 2 \text{ (mittens)} + 3 \text{ (socks)}

\[
2 + 3 + 6 + 2 + 3 = 16 \text{ hours}

\[
\text{Total time for 3 outfits} = 16 \text{ hours/outfit} \times 3 \text{ outfits} = 48 \text{ hours}
\]
